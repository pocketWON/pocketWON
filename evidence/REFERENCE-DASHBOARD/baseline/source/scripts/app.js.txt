/* Existing destinations plus isolated feature views; one live motion/view lifecycle. */
(()=>{
 const content=document.getElementById('pw-content'),context=document.getElementById('pw-header-context'),navigation=document.getElementById('pw-navigation-items');
 let currentScreen='home',currentView,currentMotion,headerFlow,featureRequest,generation=0;const memory={record:{page:1},report:{segment:'habit'}};
 const buttons=PW_TABS.filter(t=>['home','record','all'].includes(t.id)).map(tab=>{const b=PWUI.button('',()=>PWNavigation.go(tab.id),'pw-nav-item');b.dataset.screen=tab.id;b.innerHTML=pwIcon(tab.id);b.append(PWUI.el('span','',tab.label));navigation.append(b);return b;});
 function saveUI(){if(!currentView)return;const r=currentView.element;if(currentScreen==='record'&&r.dataset.page)memory.record.page=Number(r.dataset.page);if(currentScreen==='report'&&r.dataset.segment)memory.report.segment=r.dataset.segment;}
 function mountMotion(entryMotion){currentMotion?.dispose();currentMotion=null;if(currentView&&typeof PocketWONMotion!=='undefined')currentMotion=PocketWONMotion.mount(currentView.element,{entryMotion});}
 function tools(screen,route,navigate){
  const lists={record:[['영수증으로 기록하기','receipt'],['AI 추천 카테고리','category-suggestion'],['소비 전 생각 도우미','reflection'],['습관 캘린더','calendar']],goal:[['목표에 모으기 · 미리보기','goal-contribution'],['다음 용돈·목표 계획','next-plan'],['목표 관련 코칭','coaching'],['칭찬·보상 약속','rewards']],report:[['AI 습관 분석','habit-analysis'],['AI 습관 리포트','ai-report'],['부모님과 함께 보기','parent-view'],['연속 기록','streak']]};
  headerFlow=PWFeatureUI.sheet('함께 살펴보기',f=>{for(const [label,id]of lists[screen]||[])f.body.append(PWFeatureUI.action(label,()=>{f.dispose();navigate(pwFeatureRoute(id,route));},true));});
 }
 function previewControls(route,navigate){headerFlow=PWFeatureUI.sheet('개발용 Preview',f=>{f.body.append(PWFeatureUI.notice('예제 데이터는 실제 기록에 합치지 않아요. 이 모드에서 기존 금융 저장도 차단됩니다.'));const id=route.featureId||'habit-analysis',feature=PWFeatureUI.select('예시 기능',Object.values(PW_FEATURES).map(i=>[i.id,i.title]),id),state=PWFeatureUI.select('화면 상태',Object.keys(AI_FEATURE_STATE).map(s=>[s,s]),PWPreview.state(id));f.body.append(feature.element,state.element);f.footer.append(PWFeatureUI.action('이 상태 보기',()=>{PWPreview.select(feature.input.value,state.input.value);f.dispose();navigate(pwFeatureRoute(feature.input.value,route));}));});}
 function showScreen(target,moveFocus=false,entryMotion){
  const route=typeof target==='string'?{screen:target}:target,feature=route?.screen==='feature'?PW_FEATURES[route.featureId]:null,tab=PW_TABS.find(t=>t.id===route?.screen);if(!feature&&!tab)return;
  if(feature?.id==='admin'&&!PWPreview.enabled){PWNavigation.go(pwFeatureRoute('app-info'),undefined,true);return;}
  const requestGeneration=++generation;featureRequest?.abort();featureRequest=null;
  saveUI();headerFlow?.dispose();headerFlow=null;currentMotion?.dispose();currentMotion=null;currentView?.dispose();currentView=null;currentScreen=route.screen;
  const opts={...(memory[route.screen]||{}),...route},navigate=(next,motion)=>PWNavigation.go(next,motion),label=feature?.title||tab.label;document.title=label+' · 포켓WON';context.textContent=label;
  if(route.screen==='home'){context.innerHTML='Pocket<span class="pw-brand-accent">WON</span>';context.setAttribute('aria-label','포켓WON');}else context.removeAttribute('aria-label');
  context.parentElement.querySelectorAll('.pw-header-action,.pw-retry').forEach(n=>n.remove());const loaded=loadPocketWONState();
  if(route.screen==='home'&&['invalid','unavailable'].includes(loaded.status)){const retry=PWUI.button('다시',()=>navigate('home'),'pw-icon-button pw-retry');retry.setAttribute('aria-label','잔액 다시 불러오기');context.parentElement.append(retry);}
  const action=PWUI.button(PWPreview.enabled?'미리보기':route.screen==='home'?'AI 코칭':listsScreen(route.screen)?'도구':'',()=>PWPreview.enabled?previewControls(route,navigate):route.screen==='home'?navigate(pwFeatureRoute('coaching',route)):tools(route.screen,route,navigate),'pw-core-header-action pw-header-action');
  if(PWPreview.enabled||route.screen==='home'||listsScreen(route.screen)){action.setAttribute('aria-label',PWPreview.enabled?'개발용 Preview 상태 선택':route.screen==='home'?'AI 코칭 · 기능 준비 중':label+' 도구 열기');context.parentElement.append(action);}
  if(route.screen==='home')currentView=createHomeView(createHomeViewModel(loaded.state),navigate,loaded.status);
  if(route.screen==='record')currentView=createRecordView(loaded,navigate,opts);
  if(route.screen==='goal')currentView=createGoalView(loaded,navigate,opts);
  if(route.screen==='report')currentView=createReportView(loaded,navigate,opts);
  if(route.screen==='all')currentView=createAllView(navigate,opts);
  if(feature)currentView=createFeatureView(feature.id,loaded,navigate,opts);
  content.replaceChildren(currentView.element);content.scrollTop=0;mountMotion(entryMotion);currentView.mount();
  if(feature){const controller=new AbortController();featureRequest=controller;PWFeatureUI.request(feature.id,loaded,controller.signal,state=>{if(controller.signal.aborted||generation!==requestGeneration||JSON.stringify(state)===JSON.stringify(currentView?.vm))return;currentMotion?.dispose();currentMotion=null;currentView?.dispose();currentView=createFeatureView(feature.id,loaded,navigate,opts,state);content.replaceChildren(currentView.element);mountMotion();currentView.mount();});}
  const active=feature?(route.returnTo?.screen&&['home','record','all'].includes(route.returnTo.screen)?route.returnTo.screen:'all'):['home','record','all'].includes(currentScreen)?currentScreen:null;
  for(const b of buttons){if(b.dataset.screen===active)b.setAttribute('aria-current','page');else b.removeAttribute('aria-current');}
  if(moveFocus&&!document.querySelector('dialog[open]')){const view=currentView,restored=route.focusAction&&[...view.element.querySelectorAll('button[data-action]')].find(b=>b.dataset.action===route.focusAction);const h=restored||document.getElementById('pw-screen-title');const focus=()=>{if(h&&currentView===view&&h.isConnected&&!document.querySelector('dialog[open]')){if(!restored)h.tabIndex=-1;h.focus({preventScroll:!document.documentElement.classList.contains('pw-accessible')});}};if(restored)requestAnimationFrame(focus);else focus();}
 }
 function listsScreen(screen){return ['record','goal','report'].includes(screen);}
 content.addEventListener('pw:motion-refresh',event=>{if(currentView&&event.target===currentView.element)currentMotion?.refresh();});
 function accessibility(){document.documentElement.classList.toggle('pw-accessible',parseFloat(getComputedStyle(document.documentElement).fontSize)>20||innerWidth<300||innerHeight<320||(innerWidth<500&&innerHeight<480));}
 const probe=PWUI.el('span','pw-size-probe');probe.setAttribute('aria-hidden','true');document.body.append(probe);new ResizeObserver(accessibility).observe(probe);window.addEventListener('resize',accessibility);accessibility();PWNavigation.init(showScreen);
})();
