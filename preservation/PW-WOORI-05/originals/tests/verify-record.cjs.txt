async function geometry(page, sheet = false) {
  const result = await page.evaluate(sheet => {
    const content = document.querySelector('.pw-content'), dialog = document.querySelector('dialog');
    const box = node => { const r = node.getBoundingClientRect(); return { x: r.x, y: r.y, right: r.right, bottom: r.bottom, width: r.width, height: r.height }; };
    const controls = sheet ? [...dialog.querySelectorAll('button,input[type="text"],.pw-record-choice>span,.pw-record-chip>span')] : [...document.querySelectorAll('.pw-record button')];
    const fields = document.querySelector('.pw-record-fields'), footer = document.querySelector('.pw-record-sheet-footer');
    return {
      width: innerWidth, height: innerHeight, docWidth: document.documentElement.scrollWidth, contentWidth: content.clientWidth, contentScroll: content.scrollWidth,
      dialog: box(dialog), fields: box(fields), footer: box(footer), fieldsScroll: fields.scrollHeight, fieldsClient: fields.clientHeight,
      controls: controls.map(node => ({ ...box(node), scrollWidth: node.scrollWidth, clientWidth: node.clientWidth })),
      nav: box(document.querySelector('.pw-bottom-navigation')), sheetScroll: dialog.scrollWidth, sheetClient: dialog.clientWidth,
    };
  }, sheet);
  assert(result.docWidth <= result.width && result.contentScroll <= result.contentWidth, JSON.stringify(result));
  if (sheet) {
    assert(result.dialog.x >= -0.1 && result.dialog.right <= result.width + 0.1 && result.dialog.y >= -0.1 && result.dialog.bottom <= result.height + 0.1, JSON.stringify(result));
    assert(result.fields.bottom <= result.footer.y + 0.1, JSON.stringify(result));
    assert(result.footer.bottom <= result.dialog.bottom + 0.1, JSON.stringify(result));
    assert(result.sheetScroll <= result.sheetClient, JSON.stringify(result));
    assert(result.controls.every(c => c.width >= 48 && c.height >= 48 && c.x >= result.dialog.x - 0.1 && c.right <= result.dialog.right + 0.1), JSON.stringify(result));
  }
  return result;
}

async function textContrast(page, selector) {
  const values = await page.locator(selector).evaluateAll(nodes => {
    const lum = color => color.match(/[\d.]+/g).slice(0, 3).map(Number).map(v => v / 255).map(v => v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4).reduce((a, v, i) => a + v * [0.2126, 0.7152, 0.0722][i], 0);
    return nodes.filter(node => node.getBoundingClientRect().height > 0 && node.textContent).map(node => {
      let surface = node;
      while (surface.parentElement && getComputedStyle(surface).backgroundColor === 'rgba(0, 0, 0, 0)') surface = surface.parentElement;
      const a = lum(getComputedStyle(node).color), b = lum(getComputedStyle(surface).backgroundColor);
      return { text: node.textContent, ratio: (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05) };
    });
  });
  assert(values.length); assert(values.every(value => value.ratio >= 4.5), JSON.stringify(values));
  return Math.min(...values.map(value => value.ratio));
}
