function executable(engine) {
  const configured = process.env[engine === 'chromium' ? 'PW_CHROMIUM_EXECUTABLE' : 'PW_WEBKIT_EXECUTABLE'];
  if (configured) return configured;
  const cache = path.join(os.homedir(), 'Library/Caches/ms-playwright');
  const prefix = engine === 'chromium' ? 'chromium_headless_shell-' : 'webkit-';
  const versions = fs.readdirSync(cache).filter(n => n.startsWith(prefix)).sort((a, b) => b.localeCompare(a, undefined, { numeric: true }));
  assert(versions.length, `Missing ${engine} browser; set executable override`);
  return path.join(cache, versions[0], engine === 'chromium' ? 'chrome-headless-shell-mac-arm64/chrome-headless-shell' : 'pw_run.sh');
}

function observe(page, label) {
  page.on('pageerror', e => report.consoleErrors.push({ label, message: e.message }));
  page.on('console', m => { if (m.type() === 'error') report.consoleErrors.push({ label, message: m.text() }); });
  page.on('requestfailed', r => report.resourceErrors.push({ label, url: r.url(), reason: r.failure()?.errorText }));
  page.on('response', r => { if (r.status() >= 400) report.resourceErrors.push({ label, url: r.url(), status: r.status() }); });
}
async function contextFor(browser, raw, label, denied = false) {
  const context = await browser.newContext({ viewport: { width: 390, height: 844 }, locale: 'ko-KR', deviceScaleFactor: 1, colorScheme: 'light', reducedMotion: 'reduce' });
  await context.addInitScript(({ raw, denied }) => {
    const get = Storage.prototype.getItem;
    const set = Storage.prototype.setItem;
    // Test preparation only. These originals are captured before instrumentation.
    if (get.call(localStorage, 'pw-test-sentinel') === null) {
      set.call(localStorage, 'pw-test-sentinel', 'preserve');
      set.call(localStorage, 'pocketwon_theme', 'dark');
      set.call(localStorage, 'pocketwon_intro_done', '1');
      if (raw !== null) set.call(localStorage, 'pocketwon_demo_v1', raw);
    }
    window.__pwReadStorage = () => Object.fromEntries(Object.keys(localStorage).map(k => [k, get.call(localStorage, k)]));
    window.__pwBeforeStorage = window.__pwReadStorage();
    window.__pwCalls = [];
    for (const method of ['getItem', 'setItem', 'removeItem', 'clear']) {
      const original = Storage.prototype[method];
      Storage.prototype[method] = function (...args) {
        window.__pwCalls.push({ method, key: args[0] });
        if (denied && method === 'getItem') throw Error('Read unavailable');
        return original.apply(this, args);
      };
    }
  }, { raw, denied });
  const page = await context.newPage(); observe(page, label);
  await page.goto(url); await ready(page);
  return { context, page };
}
async function ready(page) {
  await page.locator('.pw-home').waitFor();
  await page.evaluate(async () => { await document.fonts.ready; await new Promise(r => requestAnimationFrame(() => requestAnimationFrame(r))); });
}
async function storageUnchanged(page) {
  const result = await page.evaluate(() => ({ before: window.__pwBeforeStorage, after: window.__pwReadStorage(), calls: window.__pwCalls }));
  assert.deepEqual(result.before, result.after);
  assert(result.calls.length > 0);
  assert(result.calls.every(c => c.method === 'getItem' && c.key === 'pocketwon_demo_v1'));
}
async function geometry(page) {
  const g = await page.evaluate(() => {
    const content = document.querySelector('.pw-content'), nav = document.querySelector('.pw-bottom-navigation');
    const home = document.querySelector('.pw-home');
    return {
      width: innerWidth, doc: document.documentElement.scrollWidth,
      content: { client: content.clientWidth, scroll: content.scrollWidth, bottom: content.getBoundingClientRect().bottom },
      navTop: nav.getBoundingClientRect().top,
      amounts: [...home.querySelectorAll('.pw-home-money')].map(el => ({ width: el.clientWidth, display: el.querySelector('.pw-home-money-display').getBoundingClientRect().width, exact: el.getAttribute('aria-label') })),
      buttons: [...document.querySelectorAll('button')].map(el => ({ name: el.textContent || el.getAttribute('aria-label'), width: el.getBoundingClientRect().width, height: el.getBoundingClientRect().height })),
      metrics: getComputedStyle(document.querySelector('.pw-home-metrics')).gridTemplateColumns,
      scheme: getComputedStyle(document.documentElement).colorScheme,
    };
  });
  assert(g.doc <= g.width, JSON.stringify(g));
  assert(g.content.scroll <= g.content.client, JSON.stringify(g));
  assert(g.content.bottom <= g.navTop + 0.05);
  assert(g.amounts.every(a => a.display <= a.width + 0.05), JSON.stringify(g.amounts));
  assert(g.buttons.every(b => b.width >= 48 && b.height >= 48), JSON.stringify(g.buttons));
  assert.equal(g.scheme, 'light');
  return g;
}

async function accessibility(page, engine) {
  assert.equal(await page.locator('h1').count(), 1);
  assert.deepEqual(await page.locator('.pw-home h2').allTextContents(), ['내 용돈', '저장된 돈 흐름', '나의 돈 습관', '지금 모으는 목표', '오늘의 미션']);
  assert.equal(await page.getByRole('progressbar', { name: '습관 점수' }).getAttribute('aria-valuenow'), '82');
  assert.equal(await page.getByRole('progressbar', { name: '목표 진행률' }).getAttribute('aria-valuenow'), '68');
  const contrast = await page.evaluate(() => {
    const luminance = color => color.match(/[\d.]+/g).slice(0, 3).map(Number).map(v => v / 255).map(v => v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4).reduce((a, v, i) => a + v * [0.2126, 0.7152, 0.0722][i], 0);
    const ratio = (a, b) => { const x = luminance(a), y = luminance(b); return (Math.max(x, y) + 0.05) / (Math.min(x, y) + 0.05); };
    return [...document.querySelectorAll('.pw-home h2, .pw-home-muted, .pw-home-period, .pw-home-money-digits, .pw-home-money-unit, .pw-home-action, .pw-home-primary, .pw-home-percent, .pw-home-score')].map(el => {
      let parent = el;
      while (parent && getComputedStyle(parent).backgroundColor === 'rgba(0, 0, 0, 0)') parent = parent.parentElement;
      return { role: el.className, ratio: ratio(getComputedStyle(el).color, getComputedStyle(parent).backgroundColor) };
    });
  });
  assert(contrast.every(c => c.ratio >= 4.5), JSON.stringify(contrast));
  if (engine === 'Chromium') fs.writeFileSync(path.join(output, 'accessibility-tree.txt'), await page.locator('#pw-content').ariaSnapshot());
  pass(`${engine}: rendered home text contrast on actual surfaces >= 4.5:1`, { minimum: Math.min(...contrast.map(c => c.ratio)) });
  await page.getByRole('button', { name: '기록하기', exact: true }).focus();
  await page.keyboard.press('Tab');
  const focus = await page.evaluate(() => ({ text: document.activeElement.textContent, visible: document.activeElement.matches(':focus-visible'), width: getComputedStyle(document.activeElement).outlineWidth }));
  assert.equal(focus.text, '리포트 보기›'); assert(focus.visible); assert.equal(focus.width, '2px');
  await page.keyboard.press('Enter'); assert.equal(await page.locator('#pw-content').innerText(), '리포트');
  await page.getByRole('button', { name: '홈', exact: true }).click(); await ready(page);
  await page.getByRole('button', { name: '기록하기', exact: true }).focus(); await page.keyboard.press('Space');
  assert.equal(await page.locator('#pw-screen-title').innerText(), '기록');
  assert.equal(await page.getByRole('button', { name: '새 기록 추가', exact: true }).count(), 1);
  await page.getByRole('button', { name: '홈', exact: true }).click(); await ready(page);
  pass(`${engine}: heading hierarchy, named progress values, focus-visible and Enter/Space activation`);
}

(async () => {
  let chrome, safari;
  try {
    adapterTests();
    chrome = await chromium.launch({ headless: true, executablePath: executable('chromium') });
    report.browsers.chromium = chrome.version();
    await matrix(chrome, 'Chromium', true); await fallbacks(chrome, 'Chromium');
    safari = await webkit.launch({ headless: true, executablePath: executable('webkit') });
    report.browsers.webkit = safari.version();

    assert.deepEqual(report.consoleErrors, []); assert.deepEqual(report.resourceErrors, []);
    pass('All isolated contexts: zero console/page errors and failed resources');
    report.status = 'PASS';
  } catch (error) { report.status = 'FAIL'; report.failure = error.stack; process.exitCode = 1; }
  finally {
    if (chrome) await chrome.close(); if (safari) await safari.close();
    report.finishedAt = new Date().toISOString();
    fs.writeFileSync(path.join(output, 'verification.json'), JSON.stringify(report, null, 2) + '\n');
    console.log(JSON.stringify({ status: report.status, checks: report.checks.length, screenshots: report.screenshots, consoleErrors: report.consoleErrors, resourceErrors: report.resourceErrors, failure: report.failure }, null, 2));
  }
})();
