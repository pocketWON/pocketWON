                document.getElementById("segOut").classList.toggle("on", recType === "out");

            input.addEventListener("input", () => {
                input.value = input.value.replace(/[^0-9]/g, "").slice(0, 8);
            });

            });

            if (receiptBtn) receiptBtn.onclick = () => {

                toast("영수증이 추가되었어요");
            };
            document.getElementById("keypad").addEventListener("click", e => {
                const b = e.target.closest("button");
                if (!b) return;
                const k = b.dataset.key;
                let v = input.value;
                if (k === "back") v = v.slice(0, -1);
                else v = (v + k).replace(/^0+(?=\d)/, "").slice(0, 8);
                input.value = v;
            });
            document.getElementById("saveBtn").onclick = () => {
                const amount = parseInt(input.value || "0", 10);
                if (!amount || amount <= 0) {
                    toast("금액을 입력해주세요");
                    return;
                }
                const tx = {
                    type: recType,
                    amount,
                    category: recType === "out" ? recCat : "용돈",
                    ts: new Date().toISOString(),
                    receipt: recReceipt
                };
                state.transactions.unshift(tx);
                if (recType === "in") {
                    state.balance += amount;
                    state.monthly.saving += amount;
                } else {
                    state.balance = Math.max(0, state.balance - amount);
                    state.monthly.spending += amount;
                }
                if (!state.recordedDays.includes(state.today)) {
                    state.recordedDays.push(state.today);
                    state.streak += 1;
                }
                saveState();
                recReceipt = false;
                toast("용돈 기록을 저장했어요! 🎉");
                setTimeout(() => renderScreen("home"), 1500);
            };
        }

        function renderCalendar() {
            const base = new Date(2024, 4, 1); // 2024년 5월
            const view = new Date(2024, 4 + state.calOffset, 1);
            const y = view.getFullYear(),
                m = view.getMonth();
            const firstDow = (new Date(y, m, 1).getDay() + 6) % 7; // 월요일 시작
            const daysIn = new Date(y, m + 1, 0).getDate();
            const isBase = state.calOffset === 0;
            const dows = ["월", "화", "수", "목", "금", "토", "일"];
            let cells = "";
            for (let i = 0; i < firstDow; i++) cells += `<div class="cal-cell"></div>`;
            for (let d = 1; d <= daysIn; d++) {
                const rec = isBase && state.recordedDays.includes(d);
                const sp = isBase && state.specialDays.includes(d);
                const today = isBase && d === state.today;
                cells += `<div class="cal-cell ${today?"today":""}">
      ${rec ? `<span class="day-ico">${sp?"🔥":"🙂"}</span>` : `<span>${d}</span>`}
      ${today ? `<span style="font-size:9.5px;font-weight:800;color:var(--on-yellow)">오늘</span>` : ""}
    </div>`;
            }
            return `
  <header class="appbar">
    <button class="icon-btn" data-back aria-label="뒤로 가기">
      <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.6" stroke-linecap="round" stroke-linejoin="round"><path d="M15 5l-7 7 7 7"/></svg>

        /* ================= THEME ================= */
        const THEME_KEY = "pocketwon_theme";
        const themeMedia = window.matchMedia("(prefers-color-scheme: dark)");

        function themePref() {
            try {
                return localStorage.getItem(THEME_KEY) || "system";
            } catch (e) {
                return "system";
            }
        }

        function isDark() {
            const p = themePref();
            return p === "dark" || (p === "system" && themeMedia.matches);
        }

        function applyTheme() {
            document.documentElement.dataset.theme = isDark() ? "dark" : "light";
        }

        function setTheme(p) {
            try {
                localStorage.setItem(THEME_KEY, p);
            } catch (e) {}
            applyTheme();
        }
        if (themeMedia.addEventListener) themeMedia.addEventListener("change", () => {
            if (themePref() === "system") applyTheme();
        });

        function themeIcon() {
            return isDark() ?
